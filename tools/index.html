Snippet LINQ (tham khảo)

              

                Cách dùng LINQ · tham khảo
                
              

            


            

              
// GET: api/Products
var query = _context.Set<Product>().AsNoTracking();

// [SỬA KHI ĐỔI ĐỔI] Thêm filter / search / sort theo ý cần:
// query = query.Where(e => e.Price > 100);
// query = query.Where(e => e.Name.Contains(keyword));
query = query.OrderBy(e => e.Name).ThenBy(e => e.Price);

var items = await query.ToListAsync();

// Nối bảng theo khóa ngoại (Include) — ví dụ: Product -> Category:
var itemsWithFk = await _context.Set<Product>()
    .Include(e => e.Category)
    .ToListAsync();

return Ok(items);
